

# kf_system

Sistema de atención al cliente web en línea en tiempo real

#install

npm i


#run


node chatserver.js


# Acceso al panel de atención al cliente


http://127.0.0.1:8080/kfhoutai.html 


# Visitar la página de prueba


http://127.0.0.1:8080/test.html 


# Integrar en la esquina inferior derecha del sitio, agregar el JS del cliente al sitio


    <!-- Incluir jQuery -->
    <script src="https://code.jquery.com/jquery-2.1.4.min.js"></script>
    <!-- Incluir Socket.io -->
    <script src="http://127.0.0.1:8080/socket.io/socket.io.js"></script>
    <!-- Incluir el JS de atención al cliente en línea -->
    <script src="http://127.0.0.1:8080/js/chat_client.js"></script>
